def get_aoai_llm_response(self, system_message, context, query):
    
        messages = []

        system_message_chat_item = {
                "role": "system",
                "content": [
                    {
                        "type": "text",
                        "text": system_message
                        }
                    ]
                }        
        messages.append(system_message_chat_item)

        user_prompt_chat_item = {
            "role": "user",
            "content": [
                    {
                        "type": "text",
                        "text": context + f'\n Query: {query}'
                    }                    
                    ]
                }   
        messages.append(user_prompt_chat_item)
            
        completion = self.aoai_client.chat.completions.create(
            model=self.aoai_deployment_name, 
            messages=messages,
            max_tokens=self.aoai_max_tokens,  
            temperature=self.aoai_temperature,  
            top_p=self.aoai_top_p,  
            frequency_penalty=0,  
            presence_penalty=0,        
            stream=False
        )  

        return completion.to_json()
